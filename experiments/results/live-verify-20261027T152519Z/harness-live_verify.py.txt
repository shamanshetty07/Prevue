def canon(v):
    return json.dumps(v, sort_keys=True, separators=(",", ":"), ensure_ascii=False)


def sha(v):
    return hashlib.sha256(canon(v).encode()).hexdigest()


def cli_validate(path):
    r = subprocess.run(["node", CLI, "validate", path], capture_output=True, text=True)
    try:
        return json.loads(r.stdout)
    except ValueError:
        return {"valid": False}

def run_session(name, prompt, tools, allowed, sentinel, turns, budget, timeout=150):
    trace = os.path.join(EVD, f"{name}.guard-trace.jsonl")
    open(trace, "w").close()
    guard = {"type": "command", "command": "python3", "args": ["-I", os.path.join(HERE, "live_guard.py"), trace, W, allowed, sentinel], "timeout": 10}
    settings = {"hooks": {"SessionStart": [{"hooks": [dict(guard)]}], "PreToolUse": [{"hooks": [{**guard, "onFailure": "block"}]}]}}
    cmd = ["claude", "-p", prompt, "--plugin-dir", ROOT, "--permission-mode", "bypassPermissions", "--tools", tools, "--disallowedTools", "mcp__claude_ai_*",
           "--max-turns", str(turns), "--max-budget-usd", str(budget), "--no-session-persistence", "--output-format", "stream-json", "--verbose",
           "--include-hook-events", "--settings", json.dumps(settings)]
    so, se = os.path.join(EVD, f"{name}.stream.jsonl"), os.path.join(EVD, f"{name}.stderr.txt")
    t0 = time.time()
    timed_out = False
    with open(so, "wb") as fo, open(se, "wb") as fe:
        p = subprocess.Popen(cmd, cwd=W, stdin=subprocess.DEVNULL, stdout=fo, stderr=fe, start_new_session=True)
        try:
            p.wait(timeout=timeout)
        except subprocess.TimeoutExpired:
            timed_out = True
        alive = bpt.kill_group(p, p.pid)
    stream = bpt.load_jsonl(so)
    init = bpt.first_init(stream) or {}
    res = next((o for o in stream if o.get("type") == "result"), {})
    uses, results = [], {}
    for o in stream:
        for c in (o.get("message", {}) or {}).get("content", []) if isinstance(o.get("message"), dict) and isinstance(o["message"].get("content"), list) else []:
            if isinstance(c, dict) and c.get("type") == "tool_use":
                uses.append({"id": c.get("id"), "name": c.get("name"), "input": c.get("input")})
            if isinstance(c, dict) and c.get("type") == "tool_result":
                txt = c.get("content")
                if isinstance(txt, list):
                    txt = " ".join(x.get("text", "") for x in txt if isinstance(x, dict))
                results[c.get("tool_use_id")] = {"is_error": bool(c.get("is_error")), "text": txt if isinstance(txt, str) else json.dumps(txt)}
    pre = [o for o in stream if o.get("type") == "system" and o.get("subtype") == "hook_response" and o.get("hook_event") == "PreToolUse"]
    ups = [o for o in stream if o.get("type") == "system" and o.get("subtype") == "hook_response" and o.get("hook_event") == "UserPromptSubmit"]
    info = {
        "name": name, "exit": p.returncode, "timed_out": timed_out, "group_alive_after": alive, "wall_s": round(time.time() - t0, 1),
        "result_subtype": res.get("subtype"), "is_error": res.get("is_error"), "num_turns": res.get("num_turns"), "duration_ms": res.get("duration_ms"),
        "total_cost_usd": res.get("total_cost_usd"), "permission_mode": init.get("permissionMode"), "tools": init.get("tools"),
        "mcp_servers": init.get("mcp_servers"), "session_start": bpt.parse_session_start(stream),
        "tool_calls": [{"name": u["name"], "input": u["input"], "result": results.get(u["id"])} for u in uses],
        "pretool": [{"class": bpt.classify_pretool(o), "reason": bpt.hook_fields(o)[4], "stderr": bpt.hook_fields(o)[1][:120]} for o in pre],
        "userpromptsubmit": [{"stdout": (o.get("stdout") or "")[:400], "exit": o.get("exit_code")} for o in ups],
        "guard_trace": bpt.load_jsonl(trace),
    }
    json.dump(info, open(os.path.join(EVD, f"{name}.summary.json"), "w"), indent=2)
    return info


def snapshot(label):
    d = {"label": label, "state_file": read_state(), "cli_status": cli_status(), "designs": design_files(), "workspace": ws_listing()}
    json.dump(d, open(os.path.join(EVD, f"state-{label}.json"), "w"), indent=2)
    return d


def reg_result(info):
    for c in info["tool_calls"]:
        if c["name"] == REG and c["result"] and not c["result"]["is_error"]:
            try:
                return json.loads(c["result"]["text"])
            except ValueError:
                return None
    return None


def verdict(checks):
    bad = [n for n, ok, kind, _ in checks if not ok and kind == "fail"]
    inc = [n for n, ok, kind, _ in checks if not ok and kind == "inc"]
    return ("FAIL" if bad else "INCONCLUSIVE" if inc else "PASS"), bad, inc


results, summaries = {}, {}


def finish(test, checks, notes=""):
    v, bad, inc = verdict(checks)
    results[test] = {"verdict": v, "failed_checks": bad, "inconclusive_checks": inc, "notes": notes,
                     "checks": [{"check": n, "ok": ok, "kind": k, "detail": d} for n, ok, k, d in checks]}
    print(f"{test}: {v}" + (f"  failed={bad}" if bad else "") + (f"  inconclusive={inc}" if inc else ""), flush=True)
    return v


def common(info, checks, expect_ss, label):
    """Checks that apply to every session: ran to completion in budget, plugin loaded, Prevue hook ran with the expected state."""
    checks += [
        (f"{label}: session completed (exit 0, no timeout, no error)", info["exit"] == 0 and not info["timed_out"] and not info["is_error"], "inc", f"exit={info['exit']} timeout={info['timed_out']} err={info['is_error']}"),
        (f"{label}: permission mode was bypassPermissions", info["permission_mode"] == "bypassPermissions", "inc", str(info["permission_mode"])),
        (f"{label}: Prevue MCP server connected", any(isinstance(s, dict) and "prevue" in str(s.get("name")) and s.get("status") == "connected" for s in (info["mcp_servers"] or [])), "inc", json.dumps(info["mcp_servers"])),
        (f"{label}: Prevue SessionStart output seen and equals expected state", bool(info["session_start"]) and all(info["session_start"].get(k) == v for k, v in expect_ss.items()), "inc", json.dumps(info["session_start"] and {k: info["session_start"].get(k) for k in ("status", "revision", "hash12")})),
    ]
